344// MANIPULATORS
345inline void MessageGUIDHashAlgo::operator()(const void* data,
346 BSLA_MAYBE_UNUSED size_t numBytes)
347{
348 // Implementation note: the implementation is based on Jon Maiga's research
349 // on different bit mixers and their qualities (look for `mxm`):
350 // https://jonkagstrom.com/bit-mixer-construction/index.html
351
352 // Typically, bit mixers are used as the last step of computing more
353 // general hashes. But it's more than enough to use it on its own for
354 // our specific use case here.
355
356 // Performance evaluation, hash quality and avalanche effect are here:
357 // https://github.com/bloomberg/blazingmq/pull/348
358
359 struct LocalFuncs {
361 inline static bsls::Types::Uint64 mix(bsls::Types::Uint64 x)
362 {
363 x *= 0xbf58476d1ce4e5b9ULL;
364 x ^= x >> 56;
365 x *= 0x94d049bb133111ebULL;
366 return x;
367 }
368
370 inline static bsls::Types::Uint64 combine(bsls::Types::Uint64 lhs,
371 bsls::Types::Uint64 rhs)
372 {
373 lhs ^= rhs + 0x517cc1b727220a95 + (lhs << 6) + (lhs >> 2);
374 return lhs;
375 }
376 };
377
378 // `data` buffer might not be aligned to 8 bytes, so recasting the pointer
379 // might lead to UB
380 bsls::Types::Uint64 parts[2];
381 bsl::memcpy(parts, data, bmqt::MessageGUID::e_SIZE_BINARY);
382
383 parts[0] = LocalFuncs::mix(parts[0]);
384 parts[1] = LocalFuncs::mix(parts[1]);
385 d_result = LocalFuncs::combine(parts[0], parts[1]);
386}
387
388inline MessageGUIDHashAlgo::result_type MessageGUIDHashAlgo::computeHash()
389{
390 return d_result;
391}
