549inline SessionOptions&
550SessionOptions::setChannelHighWatermark(bsls::Types::Int64 value)
551{
552 // PRECONDITIONS
553 BSLS_ASSERT_OPT(8 * 1024 * 1024 < value);
554 // We reserve 4MB of 'value' for control messages (see
555 // 'bmqimp::BrokerSession::k_CONTROL_DATA_WATERMARK_EXTRA'), so the
556 // data watermark ends up being 'value - 4MB'. We require 'value' to
557 // be more than double that reservation (i.e. > 8MB, not just > 4MB)
558 // so the data watermark itself keeps at least 4MB of headroom, rather
559 // than merely being positive.
560 BSLS_ASSERT_OPT(d_channelLowWatermark < value);
561
562 d_channelHighWatermark = value;
563 return *this;
564}
565
566inline SessionOptions&
567SessionOptions::setChannelWatermark(bsls::Types::Int64 lowWatermark,
568 bsls::Types::Int64 highWatermark)
569{
570 // PRECONDITIONS
571 BSLS_ASSERT_OPT(0 <= lowWatermark);
572 BSLS_ASSERT_OPT(lowWatermark < highWatermark);
573 BSLS_ASSERT_OPT(8 * 1024 * 1024 < highWatermark);
574
575 d_channelLowWatermark = lowWatermark;
576 d_channelHighWatermark = highWatermark;
577 return *this;
578}
